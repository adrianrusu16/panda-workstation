cmake_minimum_required(VERSION 3.30)
project(PandaExternalVmSingleFrame LANGUAGES CXX)
# Future VM build only; no test, install, download or launch target.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC OFF)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Gui)
find_package(KF6Svg 6.31.0 EXACT REQUIRED)
add_executable(panda-vm-single-frame single_frame_candidate.cpp)
target_link_libraries(panda-vm-single-frame PRIVATE Qt6::Gui KF6::Svg)
target_compile_options(panda-vm-single-frame PRIVATE -fPIC -O2 -Wall -Wextra -Wpedantic -Werror)
target_link_options(panda-vm-single-frame PRIVATE -Wl,--build-id=sha1 -Wl,-z,relro,-z,now,-z,noexecstack)
