cmake_minimum_required(VERSION 3.24)
project(PandaFrameProbe LANGUAGES CXX)
# Build only: no tests, installation, downloads, QML tools or post-build execution.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC OFF)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
find_package(Qt6 6.12 REQUIRED COMPONENTS Gui)
find_package(KF6Svg 6.31 REQUIRED)
add_executable(panda-frame-worker plasma_frame_worker.cpp)
target_link_libraries(panda-frame-worker PRIVATE Qt6::Gui KF6::Svg)
target_compile_options(panda-frame-worker PRIVATE -fPIC -O2 -Wall -Wextra -Wpedantic -Werror)
target_link_options(panda-frame-worker PRIVATE -Wl,--build-id=sha1)
